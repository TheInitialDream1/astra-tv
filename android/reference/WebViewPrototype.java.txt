/** Offline WebGL launcher with a small, allowlisted native API. No remote pages can use this API. */
public final class MainActivity extends Activity {
    static final String HOST="appassets.androidplatform.net", ORIGIN="https://"+HOST;
    private WebView web;
    private PermissionRequest cameraRequest;
    private ValueCallback<Uri[]> fileRequest;
    private boolean centerHeld;
    private final Map<String,byte[]> iconCache=new ConcurrentHashMap<>();
    private final Set<String> launcherPackages=new HashSet<>();
    private final Set<String> candidateAds=new HashSet<>(Arrays.asList("com.miui.analytics","com.miui.tv.analytics","com.mitv.advertise","com.mitv.shoplugin","com.duokan.videodaily","com.xm.webcontent","com.xiaomi.mitv.shop","com.miui.misightservice","com.xiaomi.tvqs"));

    @Override public void onCreate(Bundle state){
        super.onCreate(state);
        getWindow().addFlags(WindowManager.LayoutParams.FLAG_FULLSCREEN);
        getWindow().getDecorView().setSystemUiVisibility(View.SYSTEM_UI_FLAG_FULLSCREEN|View.SYSTEM_UI_FLAG_HIDE_NAVIGATION|View.SYSTEM_UI_FLAG_IMMERSIVE_STICKY|View.SYSTEM_UI_FLAG_LAYOUT_STABLE|View.SYSTEM_UI_FLAG_LAYOUT_FULLSCREEN|View.SYSTEM_UI_FLAG_LAYOUT_HIDE_NAVIGATION);
        web=new WebView(this);web.setBackgroundColor(Color.rgb(1,4,9));setContentView(web);
        WebSettings settings=web.getSettings();settings.setJavaScriptEnabled(true);settings.setDomStorageEnabled(true);settings.setMediaPlaybackRequiresUserGesture(false);settings.setAllowFileAccess(false);settings.setAllowContentAccess(true);settings.setMixedContentMode(WebSettings.MIXED_CONTENT_NEVER_ALLOW);
        settings.setSupportMultipleWindows(false);settings.setTextZoom(100);settings.setUseWideViewPort(true);settings.setLoadWithOverviewMode(true);
        CookieManager.getInstance().setAcceptThirdPartyCookies(web,false);
        WebView.setWebContentsDebuggingEnabled(BuildConfig.DEBUG);
        WebViewAssetLoader loader=new WebViewAssetLoader.Builder().setDomain(HOST).addPathHandler("/native-icons/",path->{
            String pkg=path.endsWith(".png")?path.substring(0,path.length()-4):"";
            if(!validPackage(pkg))return denied();
            try{return response("image/png",icon(pkg));}catch(Exception e){return denied();}
        }).addPathHandler("/",path->{
            if(path.contains("..")||path.startsWith("/")||path.indexOf('\\')>=0)return denied();
            String target=path.isEmpty()?"index.html":path;
            try{return new WebResourceResponse(mime(target),"UTF-8",getAssets().open("www/"+target));}catch(IOException e){return denied();}
        }).build();
        web.setWebViewClient(new WebViewClient(){
            @Override public WebResourceResponse shouldInterceptRequest(WebView view,WebResourceRequest request){
                Uri uri=request.getUrl();if(!local(uri))return denied();
                WebResourceResponse response=loader.shouldInterceptRequest(uri);return response==null?denied():response;
            }
            @Override public boolean shouldOverrideUrlLoading(WebView view,WebResourceRequest request){return !local(request.getUrl());}
            @Override public void onPageFinished(WebView view,String url){Log.i("ASTRA","Page ready: "+url);}
            @Override public boolean onRenderProcessGone(WebView view,RenderProcessGoneDetail detail){
                new AlertDialog.Builder(MainActivity.this).setTitle("图形界面已停止").setMessage("可以重新启动桌面，或先打开电视设置。").setPositiveButton("重启桌面",(d,w)->recreate()).setNegativeButton("系统设置",(d,w)->openSystem("settings")).show();return true;
            }
        });
        web.setWebChromeClient(new WebChromeClient(){
            @Override public boolean onConsoleMessage(ConsoleMessage m){Log.println(m.messageLevel()==ConsoleMessage.MessageLevel.ERROR?Log.ERROR:Log.INFO,"ASTRA_WEB",m.message()+" @ "+m.sourceId()+":"+m.lineNumber());return true;}
            @Override public void onPermissionRequest(PermissionRequest request){runOnUiThread(()->{
                if(!local(request.getOrigin())||!Arrays.asList(request.getResources()).contains(PermissionRequest.RESOURCE_VIDEO_CAPTURE)){request.deny();return;}
                if(checkSelfPermission(Manifest.permission.CAMERA)==PackageManager.PERMISSION_GRANTED)request.grant(new String[]{PermissionRequest.RESOURCE_VIDEO_CAPTURE});
                else{if(cameraRequest!=null)cameraRequest.deny();cameraRequest=request;requestPermissions(new String[]{Manifest.permission.CAMERA},71);}
            });}
            @Override public void onPermissionRequestCanceled(PermissionRequest request){if(cameraRequest==request)cameraRequest=null;}
            @Override public boolean onShowFileChooser(WebView view,ValueCallback<Uri[]> callback,FileChooserParams params){
                if(fileRequest!=null)fileRequest.onReceiveValue(null);fileRequest=callback;
                try{Intent pick=new Intent(Intent.ACTION_OPEN_DOCUMENT).addCategory(Intent.CATEGORY_OPENABLE).setType("*/*").putExtra(Intent.EXTRA_MIME_TYPES,new String[]{"video/*","audio/*"});startActivityForResult(pick,72);return true;}catch(Exception e){fileRequest=null;callback.onReceiveValue(null);Toast.makeText(MainActivity.this,"系统未提供文件选择器，可从电视媒体应用打开本地文件",Toast.LENGTH_LONG).show();return true;}
            }
        });
        web.addJavascriptInterface(new NativeBridge(),"AstraNative");
        web.loadUrl(ORIGIN+"/index.html");web.requestFocus();
    }
    static boolean local(Uri uri){return "https".equals(uri.getScheme())&&HOST.equals(uri.getHost());}
    static boolean validPackage(String pkg){return pkg!=null&&pkg.matches("[A-Za-z_][A-Za-z0-9_]*(\\.[A-Za-z0-9_]+)+");}
    static WebResourceResponse response(String mime,byte[] bytes){return new WebResourceResponse(mime,"UTF-8",new ByteArrayInputStream(bytes));}
    static WebResourceResponse denied(){return new WebResourceResponse("text/plain","UTF-8",403,"Forbidden",Collections.emptyMap(),new ByteArrayInputStream(new byte[0]));}
    static String mime(String path){if(path.endsWith(".js"))return "application/javascript";if(path.endsWith(".css"))return "text/css";if(path.endsWith(".wasm"))return "application/wasm";if(path.endsWith(".html"))return "text/html";if(path.endsWith(".json"))return "application/json";if(path.endsWith(".svg"))return "image/svg+xml";if(path.endsWith(".png"))return "image/png";if(path.endsWith(".jpg"))return "image/jpeg";return "application/octet-stream";}
    static JSONObject json(){return new JSONObject();}
    static JSONObject put(JSONObject object,String key,Object value){try{object.put(key,value);}catch(JSONException ignored){}return object;}
    static String result(boolean ok,String message){return put(put(json(),"ok",ok),"message",message).toString();}
    Intent appIntent(String pkg){PackageManager pm=getPackageManager();Intent i=pm.getLeanbackLaunchIntentForPackage(pkg);return i==null?pm.getLaunchIntentForPackage(pkg):i;}
    byte[] icon(String pkg)throws Exception{
        byte[] cached=iconCache.get(pkg);if(cached!=null)return cached;
        Drawable drawable=getPackageManager().getApplicationIcon(pkg);Bitmap bitmap=Bitmap.createBitmap(160,160,Bitmap.Config.ARGB_8888);Canvas canvas=new Canvas(bitmap);drawable.setBounds(0,0,160,160);drawable.draw(canvas);ByteArrayOutputStream stream=new ByteArrayOutputStream();bitmap.compress(Bitmap.CompressFormat.PNG,100,stream);bitmap.recycle();byte[] bytes=stream.toByteArray();iconCache.put(pkg,bytes);return bytes;
    }
    JSONArray installedApps(){
        PackageManager pm=getPackageManager();Map<String,ResolveInfo> collected=new TreeMap<>();
        for(String category:new String[]{Intent.CATEGORY_LEANBACK_LAUNCHER,Intent.CATEGORY_LAUNCHER})for(ResolveInfo info:pm.queryIntentActivities(new Intent(Intent.ACTION_MAIN).addCategory(category),0))if(!getPackageName().equals(info.activityInfo.packageName))collected.put(info.activityInfo.packageName,info);
        JSONArray array=new JSONArray();List<ResolveInfo> sorted=new ArrayList<>(collected.values());sorted.sort((a,b)->a.loadLabel(pm).toString().compareToIgnoreCase(b.loadLabel(pm).toString()));
        for(ResolveInfo info:sorted){String pkg=info.activityInfo.packageName;JSONObject a=put(put(put(put(put(json(),"id",pkg),"packageName",pkg),"name",info.loadLabel(pm).toString()),"iconUrl",ORIGIN+"/native-icons/"+pkg+".png"),"system",(info.activityInfo.applicationInfo.flags&ApplicationInfo.FLAG_SYSTEM)!=0);String lower=pkg.toLowerCase(Locale.ROOT);String group=lower.matches(".*(bilibili|iqiyi|qiyi|youku|netflix|youtube|jellyfin|emby|plex|player|tencent\\.qqlive).*" )?"视频":lower.matches(".*(music|spotify|audio).*" )?"音乐":info.activityInfo.applicationInfo.category==ApplicationInfo.CATEGORY_GAME?"游戏":"应用";put(a,"group",group);array.put(a);}
        return array;
    }
    Set<String> protectedPackages(){
        Set<String> protectedSet=new HashSet<>(Arrays.asList(getPackageName(),"android","com.android.systemui","com.android.settings","com.android.tv.settings","com.xiaomi.mitv.settings","com.mitv.tvhome","mitv.service","com.xiaomi.mitv.tvplayer","com.mi.bluetooth","com.mitv.overseaservice","com.xiaomi.mitv.tvmanager","com.xiaomi.voicecontrol","com.xiaomi.wakeupservice","com.xiaomi.miplay","com.xiaomi.mi_connect_service","com.xiaomi.mitv.smartshare","com.xiaomi.hyper_connect.tv","com.xiaomi.mitv.upgrade"));
        PackageManager pm=getPackageManager();for(ResolveInfo r:pm.queryIntentActivities(new Intent(Intent.ACTION_MAIN).addCategory(Intent.CATEGORY_HOME),0))protectedSet.add(r.activityInfo.packageName);
        try{TvInputManager manager=(TvInputManager)getSystemService(TV_INPUT_SERVICE);if(manager!=null)for(TvInputInfo info:manager.getTvInputList())protectedSet.add(info.getServiceInfo().packageName);}catch(Exception ignored){}
        for(String action:new String[]{Settings.ACTION_SETTINGS,"android.intent.action.ASSIST"})for(ResolveInfo r:pm.queryIntentActivities(new Intent(action),0))protectedSet.add(r.activityInfo.packageName);
        return protectedSet;
    }
    void start(Intent intent){runOnUiThread(()->{try{startActivity(intent);}catch(Exception e){Toast.makeText(this,"该电视不支持此入口："+e.getClass().getSimpleName(),Toast.LENGTH_LONG).show();}});}
    void openSystem(String page){String action;switch(page){case "network":action=Settings.ACTION_WIFI_SETTINGS;break;case "home":action=Settings.ACTION_HOME_SETTINGS;break;case "inputs":action="android.settings.TV_INPUT_SETTINGS";break;case "cast":action=Settings.ACTION_CAST_SETTINGS;break;case "apps":action=Settings.ACTION_MANAGE_APPLICATIONS_SETTINGS;break;default:action=Settings.ACTION_SETTINGS;}
        start(SystemSettings.intent(this,action));}

    public final class NativeBridge {
        @JavascriptInterface public String getInstalledApps(){return installedApps().toString();}
        @JavascriptInterface public String launchApp(String pkg){if(!validPackage(pkg))return result(false,"应用包名无效");Intent intent=appIntent(pkg);if(intent==null)return result(false,"没有可启动的应用入口");start(intent);return result(true,"正在打开应用");}
        @JavascriptInterface public void openSettings(String page){openSystem(page);}
        @JavascriptInterface public String appInfo(String pkg){if(!validPackage(pkg))return result(false,"应用包名无效");Intent i=SystemSettings.appInfo(MainActivity.this,pkg);if(i==null)return result(false,"系统没有提供应用详情入口");start(i);return result(true,"已打开系统应用详情");}
        @JavascriptInterface public String uninstallApp(String pkg){try{if(!validPackage(pkg)||protectedPackages().contains(pkg)||(getPackageManager().getApplicationInfo(pkg,0).flags&ApplicationInfo.FLAG_SYSTEM)!=0)return result(false,"系统或受保护的应用保留，请在系统详情中查看");start(new Intent(Intent.ACTION_DELETE,Uri.parse("package:"+pkg)));return result(true,"请在系统窗口中确认卸载");}catch(Exception e){return result(false,"无法读取此应用");}}
        @JavascriptInterface public String getInputs(){JSONArray result=new JSONArray();try{TvInputManager manager=(TvInputManager)getSystemService(TV_INPUT_SERVICE);if(manager!=null)for(TvInputInfo info:manager.getTvInputList()){if(info.isHidden(MainActivity.this)||!info.isPassthroughInput())continue;int state=manager.getInputState(info.getId());result.put(put(put(put(json(),"id",info.getId()),"name",info.loadLabel(MainActivity.this).toString()),"state",state==TvInputManager.INPUT_STATE_CONNECTED?"已连接":state==TvInputManager.INPUT_STATE_DISCONNECTED?"未连接":"待机"));}}catch(Exception e){Log.w("ASTRA","TV input listing unavailable",e);}return result.toString();}
        @JavascriptInterface public String switchInput(String id){try{TvInputManager m=(TvInputManager)getSystemService(TV_INPUT_SERVICE);TvInputInfo info=m==null?null:m.getTvInputInfo(id);if(info==null||!info.isPassthroughInput())return result(false,"未找到该信号源");start(new Intent(Intent.ACTION_VIEW,TvContract.buildChannelUriForPassthroughInput(id)));return result(true,"正在切换信号源");}catch(Exception e){return result(false,"系统未开放此信号源");}}
        @JavascriptInterface public String getDeviceInfo(){
            ActivityManager.MemoryInfo memory=new ActivityManager.MemoryInfo();((ActivityManager)getSystemService(ACTIVITY_SERVICE)).getMemoryInfo(memory);StatFs storage=new StatFs(getFilesDir().getPath());
            ConnectivityManager cm=(ConnectivityManager)getSystemService(CONNECTIVITY_SERVICE);NetworkCapabilities capabilities=cm.getNetworkCapabilities(cm.getActiveNetwork());String network=capabilities==null?"离线":capabilities.hasTransport(NetworkCapabilities.TRANSPORT_ETHERNET)?"以太网":capabilities.hasTransport(NetworkCapabilities.TRANSPORT_WIFI)?"Wi-Fi":"网络已连接";
            JSONObject o=put(put(put(put(put(put(put(json(),"model",Build.MODEL),"manufacturer",Build.MANUFACTURER),"android",Build.VERSION.RELEASE),"sdk",Build.VERSION.SDK_INT),"memoryFree",memory.availMem),"storageFree",storage.getAvailableBytes()),"network",network);
            PackageInfo provider=WebView.getCurrentWebViewPackage();put(o,"webview",provider==null?"未知":provider.versionName);return o.toString();
        }
        @JavascriptInterface public String getCleanupReport(){
            JSONArray candidates=new JSONArray();int total=0,disabled=0;Set<String> keep=protectedPackages();PackageManager pm=getPackageManager();
            for(ApplicationInfo app:pm.getInstalledApplications(0)){total++;if(!app.enabled)disabled++;if(candidateAds.contains(app.packageName)&&!keep.contains(app.packageName))candidates.put(put(put(put(json(),"id",app.packageName),"name",app.loadLabel(pm).toString()),"enabled",app.enabled));}
            return put(put(put(put(json(),"installed",total),"disabled",disabled),"candidates",candidates),"mode","adb-review-required").toString();
        }
    }
    @Override public void onRequestPermissionsResult(int code,String[] permissions,int[] results){super.onRequestPermissionsResult(code,permissions,results);if(code==71&&cameraRequest!=null){if(results.length>0&&results[0]==PackageManager.PERMISSION_GRANTED)cameraRequest.grant(new String[]{PermissionRequest.RESOURCE_VIDEO_CAPTURE});else cameraRequest.deny();cameraRequest=null;}}
    @Override protected void onActivityResult(int request,int result,Intent data){super.onActivityResult(request,result,data);if(request==72&&fileRequest!=null){fileRequest.onReceiveValue(result==RESULT_OK&&data!=null?new Uri[]{data.getData()}:null);fileRequest=null;}}
    @Override protected void onNewIntent(Intent intent){super.onNewIntent(intent);setIntent(intent);if(web!=null)web.evaluateJavascript("window.dispatchEvent(new Event('astra-native-home'))",null);}
    @Override protected void onPause(){super.onPause();if(web!=null)web.evaluateJavascript("(function stop(w){w.dispatchEvent(new Event('astra-native-pause'));for(const f of w.document.querySelectorAll('iframe')){try{stop(f.contentWindow)}catch(e){}}})(window)",ignored->{if(web!=null)web.onPause();});}
    @Override protected void onResume(){super.onResume();if(web!=null){web.onResume();web.evaluateJavascript("window.dispatchEvent(new Event('astra-native-resume'))",null);}}
    @Override protected void onDestroy(){if(cameraRequest!=null)cameraRequest.deny();if(web!=null){web.removeJavascriptInterface("AstraNative");web.destroy();web=null;}super.onDestroy();}
    @Override public boolean dispatchKeyEvent(KeyEvent event){
        int code=event.getKeyCode();String key=code==KeyEvent.KEYCODE_DPAD_UP?"ArrowUp":code==KeyEvent.KEYCODE_DPAD_DOWN?"ArrowDown":code==KeyEvent.KEYCODE_DPAD_LEFT?"ArrowLeft":code==KeyEvent.KEYCODE_DPAD_RIGHT?"ArrowRight":code==KeyEvent.KEYCODE_BACK?"Escape":code==KeyEvent.KEYCODE_MENU?"ContextMenu":code==KeyEvent.KEYCODE_DPAD_CENTER||code==KeyEvent.KEYCODE_ENTER?"Enter":null;
        if(key==null||web==null)return super.dispatchKeyEvent(event);
        if("Enter".equals(key)){if(event.getAction()==KeyEvent.ACTION_DOWN){if(event.getRepeatCount()==0)centerHeld=false;if(event.isLongPress()&&!centerHeld){centerHeld=true;web.evaluateJavascript("window.dispatchEvent(new Event('astra-native-app-menu'))",null);}return true;}if(centerHeld)return true;}
        else if(event.getAction()!=KeyEvent.ACTION_DOWN)return true;
        boolean repeat=event.getRepeatCount()>0;
        String script="(function key(w){const d=w.document,a=d.activeElement;if(a&&a.tagName==='IFRAME'){try{key(a.contentWindow);return}catch(e){}}const e=new KeyboardEvent('keydown',{key:"+JSONObject.quote(key)+",code:"+JSONObject.quote(key)+",bubbles:true,cancelable:true,repeat:"+repeat+"});const target=a||d.body;target.dispatchEvent(e);if(!e.defaultPrevented&&e.key==='Enter'&&a&&a.tagName==='BUTTON')a.click();})(window)";
        web.evaluateJavascript(script,null);return true;
    }
